cmake_minimum_required(VERSION 3.20)
project(glrmask_external_consumer LANGUAGES C CXX)
find_package(glrmask 1.0 CONFIG REQUIRED)
enable_testing()
add_executable(cpp_consumer main.cpp)
target_link_libraries(cpp_consumer PRIVATE glrmask::glrmask)
add_test(NAME external_cpp COMMAND cpp_consumer)
add_executable(c_consumer main.c)
target_link_libraries(c_consumer PRIVATE glrmask::c)
add_test(NAME external_c COMMAND c_consumer)
