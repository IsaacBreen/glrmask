cmake_minimum_required(VERSION 3.20)
project(glrmask VERSION 0.3.1 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(GLRMASK_SHARED "Use the Rust shared library instead of the static library" OFF)
option(GLRMASK_BUILD_TESTS "Build native binding regression tests" OFF)
option(GLRMASK_BUILD_EXAMPLES "Build the C++ decode example" ON)
set(GLRMASK_CARGO_PROFILE "release" CACHE STRING "Cargo build profile")
set(GLRMASK_RUST_TARGET "" CACHE STRING "Optional Rust target triple")
set(GLRMASK_LIBRARY_DIR "" CACHE PATH "Use existing native artifacts here without invoking Cargo")
set(GLRMASK_CARGO_TARGET_DIR "$ENV{CARGO_TARGET_DIR}" CACHE PATH "Owned Cargo target directory")
if(NOT GLRMASK_CARGO_TARGET_DIR)
  set(GLRMASK_CARGO_TARGET_DIR "${CMAKE_CURRENT_BINARY_DIR}/cargo-target")
endif()
get_filename_component(GLRMASK_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(GLRMASK_CARGO_PROFILE STREQUAL "dev")
  set(GLRMASK_PROFILE_DIRECTORY "debug")
else()
  set(GLRMASK_PROFILE_DIRECTORY "${GLRMASK_CARGO_PROFILE}")
endif()
if(NOT GLRMASK_LIBRARY_DIR)
  set(GLRMASK_LIBRARY_DIR "${GLRMASK_CARGO_TARGET_DIR}")
  if(GLRMASK_RUST_TARGET)
    string(APPEND GLRMASK_LIBRARY_DIR "/${GLRMASK_RUST_TARGET}")
  endif()
  string(APPEND GLRMASK_LIBRARY_DIR "/${GLRMASK_PROFILE_DIRECTORY}")
  find_program(CARGO_EXECUTABLE NAMES cargo REQUIRED)
  set(GLRMASK_CARGO_ARGS build --locked --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/Cargo.toml" --profile "${GLRMASK_CARGO_PROFILE}")
  if(GLRMASK_RUST_TARGET)
    list(APPEND GLRMASK_CARGO_ARGS --target "${GLRMASK_RUST_TARGET}")
  endif()
  set(GLRMASK_BUILD_RUST TRUE)
endif()
if(GLRMASK_SHARED)
  set(GLRMASK_KIND SHARED)
  if(WIN32)
    set(GLRMASK_FILENAME "glrmask_c.dll")
    set(GLRMASK_IMPLIB_FILENAME "glrmask_c.dll.lib")
    set(GLRMASK_INSTALL_LOCATION "${CMAKE_INSTALL_BINDIR}/${GLRMASK_FILENAME}")
  elseif(APPLE)
    set(GLRMASK_FILENAME "libglrmask_c.dylib")
  else()
    set(GLRMASK_FILENAME "libglrmask_c.so")
  endif()
else()
  set(GLRMASK_KIND STATIC)
  if(WIN32)
    set(GLRMASK_FILENAME "glrmask_c.lib")
  else()
    set(GLRMASK_FILENAME "libglrmask_c.a")
  endif()
endif()
if(NOT GLRMASK_INSTALL_LOCATION)
  set(GLRMASK_INSTALL_LOCATION "${CMAKE_INSTALL_LIBDIR}/${GLRMASK_FILENAME}")
endif()
set(GLRMASK_LIBRARY "${GLRMASK_LIBRARY_DIR}/${GLRMASK_FILENAME}")
set(GLRMASK_BYPRODUCTS "${GLRMASK_LIBRARY}")
if(GLRMASK_IMPLIB_FILENAME)
  list(APPEND GLRMASK_BYPRODUCTS "${GLRMASK_LIBRARY_DIR}/${GLRMASK_IMPLIB_FILENAME}")
endif()
if(GLRMASK_BUILD_RUST)
  add_custom_target(glrmask_rust ALL
    COMMAND "${CMAKE_COMMAND}" -E env "CARGO_TARGET_DIR=${GLRMASK_CARGO_TARGET_DIR}" "${CARGO_EXECUTABLE}" ${GLRMASK_CARGO_ARGS}
    WORKING_DIRECTORY "${GLRMASK_SOURCE_ROOT}"
    BYPRODUCTS ${GLRMASK_BYPRODUCTS}
    COMMENT "Building GLRMask native engine with Cargo"
    VERBATIM USES_TERMINAL)
elseif(NOT EXISTS "${GLRMASK_LIBRARY}")
  message(FATAL_ERROR "Native artifact does not exist: ${GLRMASK_LIBRARY}")
endif()
add_library(glrmask::c ${GLRMASK_KIND} IMPORTED GLOBAL)
set_target_properties(glrmask::c PROPERTIES
  IMPORTED_LOCATION "${GLRMASK_LIBRARY}"
  INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(GLRMASK_BUILD_RUST)
  add_dependencies(glrmask::c glrmask_rust)
endif()
if(GLRMASK_SHARED)
  set_property(TARGET glrmask::c PROPERTY INTERFACE_COMPILE_DEFINITIONS GLRMASK_SHARED)
  if(GLRMASK_IMPLIB_FILENAME)
    set_property(TARGET glrmask::c PROPERTY IMPORTED_IMPLIB "${GLRMASK_LIBRARY_DIR}/${GLRMASK_IMPLIB_FILENAME}")
  endif()
else()
  if(WIN32)
    # rustc --print native-static-libs, MSVC target. Rust and C++ free their
    # own allocations; no CRT heap allocation is freed across the boundary.
    set(GLRMASK_SYSTEM_LIBS kernel32 bcrypt advapi32 kernel32 ntdll userenv ws2_32 dbghelp)
  else()
    find_package(Threads REQUIRED)
    set(GLRMASK_SYSTEM_LIBS Threads::Threads ${CMAKE_DL_LIBS} m)
    if(APPLE)
      list(APPEND GLRMASK_SYSTEM_LIBS iconv)
    else()
      list(APPEND GLRMASK_SYSTEM_LIBS rt)
    endif()
  endif()
  set_property(TARGET glrmask::c PROPERTY INTERFACE_LINK_LIBRARIES "${GLRMASK_SYSTEM_LIBS}")
endif()
add_library(glrmask_cpp INTERFACE)
add_library(glrmask::glrmask ALIAS glrmask_cpp)
target_link_libraries(glrmask_cpp INTERFACE glrmask::c)
target_compile_features(glrmask_cpp INTERFACE cxx_std_17)
function(glrmask_runtime_for target)
  if(GLRMASK_SHARED AND WIN32)
    add_custom_command(TARGET ${target} POST_BUILD
      COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${GLRMASK_LIBRARY}" "$<TARGET_FILE_DIR:${target}>"
      VERBATIM)
  endif()
endfunction()
if(GLRMASK_BUILD_EXAMPLES)
  add_executable(glrmask_decode examples/decode.cpp)
  target_link_libraries(glrmask_decode PRIVATE glrmask::glrmask)
  glrmask_runtime_for(glrmask_decode)
endif()
if(GLRMASK_BUILD_TESTS)
  enable_testing()
  # The test executable creates std::threads itself. A shared Rust library's
  # dependencies do not supply the executable's pthread link requirements on
  # older glibc (libpthread was separate before glibc 2.34).
  find_package(Threads REQUIRED)
  add_executable(glrmask_cpp_tests tests/bindings.cpp)
  target_link_libraries(glrmask_cpp_tests PRIVATE glrmask::glrmask Threads::Threads)
  glrmask_runtime_for(glrmask_cpp_tests)
  add_test(NAME glrmask_cpp COMMAND glrmask_cpp_tests)
  add_executable(glrmask_c_tests tests/c_api.c)
  target_link_libraries(glrmask_c_tests PRIVATE glrmask::c)
  glrmask_runtime_for(glrmask_c_tests)
  add_test(NAME glrmask_c COMMAND glrmask_c_tests)
endif()
install(FILES include/glrmask.h include/glrmask.hpp DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
if(GLRMASK_SHARED AND WIN32)
  install(FILES "${GLRMASK_LIBRARY}" DESTINATION "${CMAKE_INSTALL_BINDIR}")
  install(FILES "${GLRMASK_LIBRARY_DIR}/${GLRMASK_IMPLIB_FILENAME}" DESTINATION "${CMAKE_INSTALL_LIBDIR}")
else()
  install(FILES "${GLRMASK_LIBRARY}" DESTINATION "${CMAKE_INSTALL_LIBDIR}")
endif()
install(FILES README.md "${GLRMASK_SOURCE_ROOT}/LICENSE-MIT" "${GLRMASK_SOURCE_ROOT}/LICENSE-APACHE" "${GLRMASK_SOURCE_ROOT}/crates/glrmask-grammar/NOTICE-LARK" DESTINATION "${CMAKE_INSTALL_DOCDIR}")
install(DIRECTORY examples/ DESTINATION "${CMAKE_INSTALL_DOCDIR}/examples")
configure_package_config_file(cmake/glrmaskConfig.cmake.in
  "${CMAKE_CURRENT_BINARY_DIR}/glrmaskConfig.cmake"
  INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/glrmask")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/glrmaskConfigVersion.cmake"
  VERSION "${PROJECT_VERSION}" COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/glrmaskConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/glrmaskConfigVersion.cmake"
  DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/glrmask")
